juce_add_plugin(SootheClone
    COMPANY_NAME "YJ Audio"
    PLUGIN_MANUFACTURER_CODE Yuit
    PLUGIN_CODE Sthc
    FORMATS VST3 Standalone
    PRODUCT_NAME "I Soothea"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(SootheClone PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_features(SootheClone PUBLIC cxx_std_17)
target_compile_definitions(SootheClone PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SootheClone PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
    target_compile_options(SootheClone PRIVATE /utf-8)
endif()
